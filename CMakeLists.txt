cmake_minimum_required(VERSION 3.21)
project(SIMRA VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)
find_package(Threads REQUIRED)
add_library(simra_core src/core.cpp src/json.cpp)
target_include_directories(simra_core PUBLIC include)
target_link_libraries(simra_core PUBLIC Qt6::Core Threads::Threads)
target_compile_options(simra_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(simra src/main.cpp src/window.cpp)
target_link_libraries(simra PRIVATE simra_core Qt6::Widgets)
target_compile_definitions(simra PRIVATE SIMRA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
enable_testing()
add_executable(simra_tests tests/tests.cpp)
target_link_libraries(simra_tests PRIVATE simra_core)
target_compile_definitions(simra_tests PRIVATE SIMRA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME simra_tests COMMAND simra_tests)
